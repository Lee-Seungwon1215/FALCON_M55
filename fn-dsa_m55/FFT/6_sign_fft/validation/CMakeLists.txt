cmake_minimum_required(VERSION 3.20.0)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(COMMON "${WORKSPACE}/fn-dsa_m55/measurement_mlkem_native")
set(FIXTURES "${WORKSPACE}/fn-dsa_m55/sep_27_ntrusolve_fft/validation")
get_filename_component(CRYPTO "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
# BASELINE is a test-only, separate firmware build; production has no switch.
if(BASELINE)
  set(CRYPTO "${WORKSPACE}/Final_code/Before_slothy")
endif()
if(ORIGINAL)
  set(CRYPTO "${WORKSPACE}/fn-dsa_m55/M55_ref")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(BOARD_LINKER_SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/fndsa_dtcm_linker.ld")
execute_process(COMMAND ${Python3_EXECUTABLE} "${COMMON}/app/generate_dtcm_linker.py"
  "$ENV{ZEPHYR_BASE}/include/zephyr/arch/arm/cortex_m/scripts/linker.ld"
  "${BOARD_LINKER_SCRIPT}" COMMAND_ERROR_IS_FATAL ANY)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(sign_fft_fp64_validation C ASM)
set(C_NAMES codec mq sha3 sysrng util kgen kgen_fxp kgen_gauss kgen_mp31
  kgen_ntru kgen_poly kgen_zint31 kgen_fft_mve kgen_fft_bridge
  sign sign_core sign_fpoly sign_fpr sign_sampler vrfy)
set(S_NAMES codec_cm4 sha3_cm4 sign_fpr_cm4 sign_sampler_cm4
  mq_cm55 kgen_mp31_cm55 kgen_fft_cm55)
if(NOT BASELINE)
  list(APPEND S_NAMES sign_ldl_cm55 sign_split_merge_cm55)
endif()
if(NOT MODE STREQUAL "sign_native")
  list(APPEND S_NAMES sign_fft_cm55)
endif()
if(ORIGINAL)
  if(NOT MODE STREQUAL "sign")
    message(FATAL_ERROR "Original reference comparator is a signing-only test")
  endif()
  list(REMOVE_ITEM C_NAMES kgen_fft_mve kgen_fft_bridge)
  set(S_NAMES codec_cm4 sha3_cm4 sign_fpr_cm4 sign_sampler_cm4 mq_cm4)
  # Test-only: enable the original reference's assembly on this M55 board.
  target_compile_definitions(app PRIVATE FNDSA_ASM_CORTEXM4=1 FNDSA_ASM_CORTEXM55=1)
endif()
if(MODE STREQUAL "sign_detail" OR MODE STREQUAL "sign_control")
  if(BASELINE)
    message(FATAL_ERROR "Signing profile measures the local production tree only")
  endif()
  set(SIGN_PROFILE_DIR "${CMAKE_CURRENT_BINARY_DIR}/sign_generated")
  execute_process(COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/sign_profile/generate.py"
    "${SIGN_PROFILE_DIR}" "${MODE}" "${CRYPTO}"
    COMMAND_ERROR_IS_FATAL ANY)
endif()
foreach(name IN LISTS C_NAMES)
  if(DEFINED SIGN_PROFILE_DIR AND name MATCHES "^(sign|sign_core|sign_sampler|sign_fpoly)$")
    list(APPEND CRYPTO_FILES "${SIGN_PROFILE_DIR}/${name}.c")
  else()
    list(APPEND CRYPTO_FILES "${CRYPTO}/${name}.c")
  endif()
endforeach()
foreach(name IN LISTS S_NAMES)
  if(DEFINED SIGN_PROFILE_DIR AND name STREQUAL "sign_sampler_cm4")
    list(APPEND CRYPTO_FILES "${SIGN_PROFILE_DIR}/${name}.s")
  else()
    list(APPEND CRYPTO_FILES "${CRYPTO}/${name}.s")
  endif()
endforeach()
if(MODE STREQUAL "kernel" OR MODE STREQUAL "sign_native")
  execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/generate_oracle.py"
    "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
endif()
if(DEFINED SIGN_PROFILE_DIR)
  set(BENCH "${SIGN_PROFILE_DIR}/benchmark.c" "${SIGN_PROFILE_DIR}/profile.c")
  if(MODE STREQUAL "sign_detail")
    list(APPEND BENCH "${SIGN_PROFILE_DIR}/fft_probes.c")
  endif()
  target_include_directories(app PRIVATE "${SIGN_PROFILE_DIR}")
elseif(MODE STREQUAL "ldl")
  set(BENCH "${CMAKE_CURRENT_SOURCE_DIR}/ldl_kernel.c" "${CMAKE_CURRENT_SOURCE_DIR}/oracle_ldl.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/native_ldl.c" "${CMAKE_CURRENT_SOURCE_DIR}/poly_abi_probe.s")
elseif(MODE STREQUAL "poly")
  set(BENCH "${CMAKE_CURRENT_SOURCE_DIR}/split_merge_kernel.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/oracle_split_merge.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/native_split_merge.c" "${CMAKE_CURRENT_SOURCE_DIR}/poly_abi_probe.s")
elseif(MODE STREQUAL "kernel")
  set(BENCH "${CMAKE_CURRENT_SOURCE_DIR}/kernel.c" "${CMAKE_CURRENT_BINARY_DIR}/oracle.c"
    "${CMAKE_CURRENT_BINARY_DIR}/native_c.c" "${CMAKE_CURRENT_SOURCE_DIR}/abi_probe.s")
elseif(MODE STREQUAL "sign_native")
  # Test-only frozen C comparator, not a selectable production backend.
  set(BENCH "${CMAKE_CURRENT_SOURCE_DIR}/sign_perf.c" "${CMAKE_CURRENT_BINARY_DIR}/native_c_sign.c")
elseif(MODE STREQUAL "sign")
  set(BENCH "${CMAKE_CURRENT_SOURCE_DIR}/sign_perf.c")
elseif(MODE STREQUAL "sigkat")
  set(BENCH "${FIXTURES}/board_signkat.c")
elseif(MODE STREQUAL "kat")
  set(BENCH "${FIXTURES}/board_kat.c")
elseif(MODE STREQUAL "extra")
  set(BENCH "${FIXTURES}/board_extra_kat.c")
elseif(MODE STREQUAL "api")
  set(BENCH "${FIXTURES}/security/api_audit.c")
else()
  message(FATAL_ERROR "Unknown test MODE")
endif()
set(SHIM "$ENV{MLKEM_NATIVE_PINNED_ROOT}/test/zephyr/app/shim_nucleo_n657x0_q.c")
target_sources(app PRIVATE ${CRYPTO_FILES} ${BENCH} ${SHIM} "${COMMON}/app/dtcm_startup.s")
target_include_directories(app PRIVATE "${CRYPTO}" "${FIXTURES}" "${FIXTURES}/generated")
target_compile_definitions(app PRIVATE AUDIT_BOARD=1 printf=__wrap_printf
  fprintf=__wrap_fprintf puts=__wrap_puts putchar=__wrap_putchar
  fflush=__wrap_fflush exit=__wrap_exit)
zephyr_link_libraries(-Wl,--wrap=soc_early_reset_hook -Wl,--wrap=printf
  -Wl,--wrap=fprintf -Wl,--wrap=puts -Wl,--wrap=putchar -Wl,--wrap=fflush -Wl,--wrap=exit)
zephyr_compile_definitions(USER_VECT_TAB_ADDRESS VECT_TAB_BASE_ADDRESS=0x10000000 VECT_TAB_OFFSET=0x0)
set_property(SOURCE ${CRYPTO_FILES} ${BENCH} ${SHIM} APPEND PROPERTY
  COMPILE_OPTIONS -O3 -mfpu=fpv5-d16 -ffp-contract=off -fno-fast-math -fno-strict-aliasing)
