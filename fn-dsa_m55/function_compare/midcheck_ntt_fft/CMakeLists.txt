cmake_minimum_required(VERSION 3.20.0)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(COMMON "${WORKSPACE}/fn-dsa_m55/measurement_mlkem_native")
set(CRYPTO "${CMAKE_CURRENT_SOURCE_DIR}/variants/${VARIANT}")
if(NOT VARIANT MATCHES "^(ref|ntt|full)$" OR NOT MODE MATCHES "^(perf|profile)$")
  message(FATAL_ERROR "Invalid experiment")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(BOARD_LINKER_SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/fndsa_dtcm_linker.ld")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/generate_linker.py"
  "$ENV{ZEPHYR_BASE}/include/zephyr/arch/arm/cortex_m/scripts/linker.ld"
  "${BOARD_LINKER_SCRIPT}" "${MODE}" COMMAND_ERROR_IS_FATAL ANY)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(midcheck C ASM)
set(C_NAMES codec mq sha3 sysrng util kgen kgen_fxp kgen_gauss kgen_mp31 kgen_ntru
  kgen_poly kgen_zint31 sign sign_core sign_fpoly sign_fpr sign_sampler vrfy)
if(VARIANT STREQUAL "full")
  list(APPEND C_NAMES kgen_fft_mve kgen_fft_bridge)
endif()
set(BENCH "${CMAKE_CURRENT_SOURCE_DIR}/benchmark.c")
if(MODE STREQUAL "profile")
  if(NOT VARIANT STREQUAL "full")
    message(FATAL_ERROR "Profile current integrated implementation only")
  endif()
  set(PROBE "${CMAKE_CURRENT_BINARY_DIR}/instrumented")
  execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/instrument.py"
    "${CRYPTO}" "${PROBE}" COMMAND_ERROR_IS_FATAL ANY)
  set(BENCH "${PROBE}/benchmark.c" "${PROBE}/probe.c" "${PROBE}/sign_probes.c")
  target_include_directories(app PRIVATE "${PROBE}")
endif()
foreach(name IN LISTS C_NAMES)
  if(DEFINED PROBE AND EXISTS "${PROBE}/${name}.c")
    list(APPEND CRYPTO_FILES "${PROBE}/${name}.c")
  else()
    list(APPEND CRYPTO_FILES "${CRYPTO}/${name}.c")
  endif()
endforeach()
file(GLOB ASM_FILES "${CRYPTO}/*.s")
foreach(path IN LISTS ASM_FILES)
  get_filename_component(name "${path}" NAME)
  if(DEFINED PROBE AND EXISTS "${PROBE}/${name}")
    list(APPEND CRYPTO_FILES "${PROBE}/${name}")
  else()
    list(APPEND CRYPTO_FILES "${path}")
  endif()
endforeach()
set(SHIM "$ENV{MLKEM_NATIVE_PINNED_ROOT}/test/zephyr/app/shim_nucleo_n657x0_q.c")
target_sources(app PRIVATE ${CRYPTO_FILES} ${BENCH} ${SHIM} "${COMMON}/app/dtcm_startup.s")
target_include_directories(app PRIVATE "${CRYPTO}")
target_compile_definitions(app PRIVATE AUDIT_BOARD=1 printf=__wrap_printf fprintf=__wrap_fprintf
  puts=__wrap_puts putchar=__wrap_putchar fflush=__wrap_fflush exit=__wrap_exit)
zephyr_link_libraries(-Wl,--wrap=soc_early_reset_hook -Wl,--wrap=printf -Wl,--wrap=fprintf
  -Wl,--wrap=puts -Wl,--wrap=putchar -Wl,--wrap=fflush -Wl,--wrap=exit)
zephyr_compile_definitions(USER_VECT_TAB_ADDRESS VECT_TAB_BASE_ADDRESS=0x10000000 VECT_TAB_OFFSET=0x0)
set_property(SOURCE ${CRYPTO_FILES} ${BENCH} ${SHIM} APPEND PROPERTY
  COMPILE_OPTIONS -O3 -mfpu=fpv5-d16 -ffp-contract=off -fno-fast-math -fno-strict-aliasing)
